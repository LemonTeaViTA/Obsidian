```java
// ExecutionPlan.java
public List<List<TaskNode>> topologicalSort() {
    // DFS 环检测
    detectCycles();
    
    // Kahn 算法分层
    Queue<TaskNode> queue = new LinkedList<>();
    // 入度为 0 的节点入队
    for (TaskNode node : nodes) {
        if (node.getInDegree() == 0) {
            queue.offer(node);
        }
    }
    
    List<List<TaskNode>> batches = new ArrayList<>();
    while (!queue.isEmpty()) {
        List<TaskNode> currentBatch = new ArrayList<>(queue);
        batches.add(currentBatch);
        queue.clear();
        
        // 处理当前批次的出边
        for (TaskNode node : currentBatch) {
            for (TaskNode dependent : node.getDependents()) {
                dependent.decreaseInDegree();
                if (dependent.getInDegree() == 0) {
                    queue.offer(dependent);
                }
            }
        }
    }
    
    return batches;
}
```

public List<MemoryEntry> retrieve(String query, int topK) {
    // 1. 查询分词
    List<String> tokens = tokenizer.tokenize(query);
    
    // 2. 计算相似度
    List<ScoredEntry> scored = new ArrayList<>();
    for (MemoryEntry entry : allMemories) {
        double score = calculateSimilarity(tokens, entry);
        scored.add(new ScoredEntry(entry, score));
    }
    
    // 3. 排序取 topK
    scored.sort(Comparator.comparingDouble(ScoredEntry::getScore).reversed());
    return scored.stream()
        .limit(topK)
        .map(ScoredEntry::getEntry)
        .collect(Collectors.toList());
}

// PathGuard.java
public Path validatePath(String userPath, Path projectRoot) {
    Path resolved = projectRoot.resolve(userPath).normalize();
    
    // 检查是否在项目根之内
    if (!resolved.startsWith(projectRoot)) {
        throw new SecurityException("Path escapes project root: " + userPath);
    }
    
    // 检查符号链接
    if (Files.isSymbolicLink(resolved)) {
        Path target = Files.readSymbolicLink(resolved);
        if (!target.startsWith(projectRoot)) {
            throw new SecurityException("Symlink escapes project root: " + userPath);
        }
    }
    
    return resolved;
}
```

// SkillRegistry.java
public void loadSkills() {
    // 三层加载位置（按优先级）
    List<Path> locations = List.of(
        resourcesPath("skills"),           // jar 内置
        userHome(".paicli/skills"),        // 用户级
        projectRoot(".paicli/skills")      // 项目级
    );
    
    for (Path location : locations) {
        if (!Files.exists(location)) continue;
        
        Files.list(location)
            .filter(Files::isDirectory)
            .forEach(skillDir -> {
                String name = skillDir.getFileName().toString();
                Skill skill = skillLoader.load(skillDir);
                
                // 后者覆盖前者
                skills.put(name, skill);
            });
    }
}
```

        INSERT INTO tasks (id, content, status, created_at)
        VALUES (?, ?, 'ENQUEUED', ?)
        """;
    
    jdbcTemplate.update(sql, 
        UUID.randomUUID().toString(),
        taskContent,
        LocalDateTime.now()
    );

public Optional<Task> dequeue() {
    // 原子操作：查询并锁定
    String sql = """
        UPDATE tasks
        SET status = 'RUNNING', started_at = ?
        WHERE id = (
            SELECT id FROM tasks
            WHERE status = 'ENQUEUED'
            ORDER BY created_at ASC
            LIMIT 1
        )
        RETURNING *
        """;

        LocalDateTime.now(),

}
