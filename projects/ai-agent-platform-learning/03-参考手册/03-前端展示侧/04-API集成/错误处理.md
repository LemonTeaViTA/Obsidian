# 错误处理

## 概述

完善的错误处理机制是保证应用稳定性和用户体验的关键。AI Agent Platform 采用多层次的错误处理策略，包括API错误处理、React错误边界、全局错误拦截、用户友好提示等，确保错误被正确捕获、记录和展示。

## 错误处理层次

```
用户操作
  ↓
组件层错误处理（try-catch、错误边界）
  ↓
API层错误处理（Axios拦截器）
  ↓
全局错误处理（window.onerror）
  ↓
错误日志记录
  ↓
用户友好提示（Toast、Modal）
```

## API错误处理

### 1. Axios响应拦截器

全局拦截API错误并统一处理。

```typescript
// src/api/open-hands-axios.ts
import axios, { AxiosError, AxiosResponse } from "axios";
import { displayErrorToast } from "#/utils/custom-toast-handlers";

export const openHands = axios.create({
  baseURL: `${window.location.protocol}//${import.meta.env.VITE_BACKEND_BASE_URL}`,
  withCredentials: true,
  timeout: 30000,
});

// 响应拦截器
openHands.interceptors.response.use(
  (response: AxiosResponse) => {
    // 检查业务逻辑错误（success: false）
    if (
      response.data &&
      typeof response.data === "object" &&
      "success" in response.data
    ) {
      if (response.data.success === false) {
        const errorMessage = response.data.message || "操作失败";
        displayErrorToast(errorMessage);

        // 创建错误对象并拒绝Promise
        const businessError = new Error(errorMessage);
        (businessError as any).response = response;
        (businessError as any).isBusinessError = true;
        return Promise.reject(businessError);
      }
    }

    return response;
  },
  (error: AxiosError) => {
    // HTTP错误处理
    if (error.response) {
      // 服务器返回错误响应
      const status = error.response.status;
      const data = error.response.data as any;

      switch (status) {
        case 400:
          displayErrorToast(data?.message || "请求参数错误");
          break;
        case 401:
          displayErrorToast("未授权，请重新登录");
          // 跳转到登录页
          window.location.href = "/login";
          break;
        case 403:
          // 检查是否是邮箱验证错误
          if (checkForEmailVerificationError(data)) {
            if (window.location.pathname !== "/settings/user") {
              window.location.reload();
            }
          } else {
            displayErrorToast("权限不足");
          }
          break;
        case 404:
          displayErrorToast("请求的资源不存在");
          break;
        case 500:
          displayErrorToast("服务器内部错误");
          break;
        case 502:
        case 503:
          displayErrorToast("服务暂时不可用，请稍后重试");
          break;
        default:
          displayErrorToast(data?.message || `请求失败 (${status})`);
      }
    } else if (error.request) {
      // 请求已发出但没有收到响应
      displayErrorToast("网络连接失败，请检查网络");
    } else {
      // 请求配置出错
      displayErrorToast("请求配置错误");
    }

    return Promise.reject(error);
  }
);

// 检查邮箱验证错误
function checkForEmailVerificationError(data: any): boolean {
  const EMAIL_NOT_VERIFIED = "EmailNotVerifiedError";

  if (typeof data === "string") {
    return data.includes(EMAIL_NOT_VERIFIED);
  }

  if (typeof data === "object" && data !== null) {
    if ("message" in data) {
      const { message } = data;
      if (typeof message === "string") {
        return message.includes(EMAIL_NOT_VERIFIED);
      }
      if (Array.isArray(message)) {
        return message.some(
          (msg) => typeof msg === "string" && msg.includes(EMAIL_NOT_VERIFIED)
        );
      }
    }

    return Object.values(data).some(
      (value) =>
        (typeof value === "string" && value.includes(EMAIL_NOT_VERIFIED)) ||
        (Array.isArray(value) &&
          value.some((v) => typeof v === "string" && v.includes(EMAIL_NOT_VERIFIED)))
    );
  }

  return false;
}
```

### 2. API函数错误处理

在API函数中进行错误检查和转换。

```typescript
// src/api/task-service/task-service.api.ts
import axios from 'axios';
import { getApiBaseUrl } from '#/utils/api-base-url';

export async function getTaskDetail(taskId: string): Promise<TaskDetail> {
  try {
    const apiBaseUrl = getApiBaseUrl();
    const response = await axios.get(
      `${apiBaseUrl}/api/taskInfo/detail`,
      { params: { id: taskId } }
    );

    // 检查业务成功标志
    if (!response.data.success) {
      throw new Error(response.data.message || '获取任务详情失败');
    }

    return response.data.data;
  } catch (error) {
    // 转换错误对象，提供更友好的错误信息
    if (axios.isAxiosError(error)) {
      if (error.response?.status === 404) {
        throw new Error('任务不存在');
      }
      if (error.response?.status === 403) {
        throw new Error('无权访问此任务');
      }
    }
    
    // 重新抛出错误，由调用方处理
    throw error;
  }
}
```

### 3. TanStack Query错误处理

在React Query中统一处理查询和变更错误。

```typescript
// src/hooks/query/use-task-detail.ts
import { useQuery } from '@tanstack/react-query';
import { getTaskDetail } from '#/api/task-service/task-service.api';

export function useTaskDetail(taskId: string | undefined) {
  return useQuery({
    queryKey: ['task-detail', taskId],
    queryFn: async () => {
      if (!taskId) {
        throw new Error('任务ID不能为空');
      }
      return getTaskDetail(taskId);
    },
    enabled: !!taskId,
    retry: (failureCount, error) => {
      // 对于404错误不重试
      if (error.message.includes('任务不存在')) {
        return false;
      }
      // 最多重试2次
      return failureCount < 2;
    },
    retryDelay: (attemptIndex) => Math.min(1000 * 2 ** attemptIndex, 30000),
  });
}
```

```typescript
import { useMutation, useQueryClient } from '@tanstack/react-query';

        displayErrorToast('任务不存在或已被删除');
      } else {
        displayErrorToast(error.message || '删除任务失败');
      }
    },
  });
}
```

## React错误边界

### 1. 全局错误边界

捕获组件树中的渲染错误。

```typescript
// src/components/shared/error-boundary.tsx
import React, { Component, ErrorInfo, ReactNode } from 'react';
import { Button } from '@heroui/react';
import { AlertTriangle } from 'lucide-react';

interface Props {
  children: ReactNode;
  fallback?: ReactNode;
}

interface State {
  hasError: boolean;
  error?: Error;
  errorInfo?: ErrorInfo;
}

export class ErrorBoundary extends Component<Props, State> {
  constructor(props: Props) {
    super(props);
    this.state = { hasError: false };
  }

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: ErrorInfo) {
    // 记录错误到日志服务
    console.error('ErrorBoundary caught an error:', error, errorInfo);
    
    // 可以发送到错误监控服务（如Sentry）
    // logErrorToService(error, errorInfo);
    
    this.setState({ error, errorInfo });
  }

  handleReset = () => {
    this.setState({ hasError: false, error: undefined, errorInfo: undefined });
  };

  render() {
    if (this.state.hasError) {
      // 如果提供了自定义fallback，使用自定义UI
      if (this.props.fallback) {
        return this.props.fallback;
      }

      // 默认错误UI
      return (
        <div className="flex flex-col items-center justify-center min-h-screen p-8">
          <AlertTriangle size={64} className="text-red-500 mb-4" />
          <h1 className="text-2xl font-bold mb-2">出错了</h1>
          <p className="text-default-500 mb-4 text-center max-w-md">
            应用遇到了一个错误。我们已记录此问题，请稍后重试。
          </p>
          
          {process.env.NODE_ENV === 'development' && this.state.error && (
            <details className="mb-4 max-w-2xl">
              <summary className="cursor-pointer text-sm text-default-500">
                查看错误详情
              </summary>
              <pre className="mt-2 p-4 bg-[#1E1E1E] rounded text-xs overflow-auto">
                {this.state.error.toString()}
                {this.state.errorInfo?.componentStack}
              </pre>
            </details>
          )}
          
          <div className="flex gap-2">
            <Button color="primary" onClick={this.handleReset}>
              重试
            </Button>
            <Button onClick={() => window.location.href = '/'}>
              返回首页
            </Button>
          </div>
        </div>
      );
    }

    return this.props.children;
  }
}
```

### 1. try-catch错误处理

```typescript
// src/components/features/home/simple-task-form.tsx
function SimpleTaskForm() {
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    
    setIsSubmitting(true);
    setError(null);

    try {
      // 表单验证
      if (!validateForm()) {
        throw new Error('请填写所有必填字段');
      }

      // 提交任务
      const response = await axios.post(url, data);

      if (!response.data.success) {
        throw new Error(response.data.message || '创建任务失败');
      }

      displaySuccessToast('任务创建成功');
      navigate(`/conversations/${response.data.data}`);
    } catch (error) {
      console.error('Submit error:', error);
      
      let errorMessage = '创建任务失败';
      
      if (axios.isAxiosError(error)) {
        if (error.response) {
          errorMessage = error.response.data?.message || 
            `服务器错误 (${error.response.status})`;
        } else if (error.request) {
          errorMessage = '网络连接失败，请检查网络';
        } else {
          errorMessage = '请求配置错误';
        }
      } else if (error instanceof Error) {
        errorMessage = error.message;
      }

      setError(errorMessage);
      displayErrorToast(errorMessage);
    } finally {
      setIsSubmitting(false);
    }
  };

  return (
    <form onSubmit={handleSubmit}>
      {/* 表单字段 */}
      
      {error && (
        <div className="p-3 bg-red-500/10 border border-red-500 rounded-lg text-red-500 text-sm">
          {error}
        </div>
      )}
      
      <Button
        type="submit"
        isLoading={isSubmitting}
        isDisabled={isSubmitting}
      >
        {isSubmitting ? '创建中...' : '创建任务'}
      </Button>
    </form>
  );
}
```

```typescript
// src/hooks/use-confirm.ts
import { Modal } from 'antd';
import { ExclamationCircleOutlined } from '@ant-design/icons';

export function useConfirm() {
  const confirm = (options: {
    title: string;
    content: string;
    onOk: () => void | Promise<void>;
    okText?: string;
    cancelText?: string;
    danger?: boolean;
  }) => {
    Modal.confirm({
      title: options.title,
      icon: <ExclamationCircleOutlined />,
      content: options.content,
      okText: options.okText || '确定',
      cancelText: options.cancelText || '取消',
      okButtonProps: { danger: options.danger },
      onOk: options.onOk,
    });
  };

  return { confirm };
}

// 使用示例
function DeleteButton({ taskId }: { taskId: number }) {
  const { confirm } = useConfirm();

    confirm({
      title: '确认删除',
      content: '确认删除该任务吗？此操作不可恢复！',
      danger: true,
      onOk: async () => {
        await deleteTaskMutation.mutateAsync(taskId);
      },
    });
  };

  return <Button onClick={handleDelete}>删除</Button>;
}
```

```typescript
interface ErrorLog {
  message: string;
  stack?: string;
  componentStack?: string;
  url: string;
  userAgent: string;

export function logError(error: Error, errorInfo?: any) {
  const errorLog: ErrorLog = {
    message: error.message,
    stack: error.stack,
    componentStack: errorInfo?.componentStack,
    url: window.location.href,
    userAgent: navigator.userAgent,
    timestamp: new Date().toISOString(),
  };

  // 发送到日志服务

}

});
```

### 3. 重试机制

```typescript
async function fetchWithRetry<T>(
  fn: () => Promise<T>,
  retries: number = 3,
  delay: number = 1000
): Promise<T> {
  try {
    return await fn();
  } catch (error) {
    if (retries === 0) throw error;
    
    await new Promise(resolve => setTimeout(resolve, delay));
    return fetchWithRetry(fn, retries - 1, delay * 2);
  }
}

// 使用
const data = await fetchWithRetry(() => getTaskDetail(taskId));
```

```typescript
const shownErrors = new Set<string>();

export function displayErrorToast(message: string) {
  if (shownErrors.has(message)) {
    return;
  }
  
  shownErrors.add(message);
```

```typescript
import { useQueryClient } from '@tanstack/react-query';

function Component() {
  const queryClient = useQueryClient();

  const handleBatchDelete = async (ids: number[]) => {
    const results = await Promise.allSettled(
      ids.map(id => deleteTask(id))
    );

    const succeeded = results.filter(r => r.status === 'fulfilled').length;
    const failed = results.filter(r => r.status === 'rejected').length;

    if (failed > 0) {
      displayErrorToast(`${failed} 个任务删除失败`);
    }
    if (succeeded > 0) {
      displaySuccessToast(`成功删除 ${succeeded} 个任务`);
      queryClient.invalidateQueries({ queryKey: ['tasks'] });
    }
  };
}
```
